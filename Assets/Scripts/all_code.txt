using UnityEngine;

public class AbilityManager : MonoBehaviour
{
    [Header("Loadout Slots")]
    [SerializeField] private BaseAbility _equippedMobility;
    [SerializeField] private BaseAbility _equippedAttack;

    private void Start()
    {
        ICharacterPhysics owner = GetComponentInParent<ICharacterPhysics>();

        if (owner == null)
        {
            Debug.LogError($"[AbilityManager] No ICharacterPhysics found on {gameObject.name} or its parents.", this);
            return;
        }

        if (_equippedMobility != null)
            _equippedMobility.Initialize(owner);

        if (_equippedAttack != null)
            _equippedAttack.Initialize(owner);
    }

    {
        _equippedMobility?.Execute(direction);
    }

    public void TryUseFire(Vector3 direction)
    {
        _equippedAttack?.Execute(direction);
    }
}

using UnityEngine;

public abstract class BaseAbility : MonoBehaviour
{
    [SerializeField] protected float _cooldownTime = 1f;

    protected float CurrentCooldown { get; private set; }
    protected ICharacterPhysics Owner { get; private set; }

    
    public void Initialize(ICharacterPhysics owner)
    {
        Owner = owner;
    }

    public abstract void Execute(Vector3 direction);    

    protected void StartCooldown()
    {
        CurrentCooldown = _cooldownTime;
    }

    protected bool IsReady => CurrentCooldown <= 0f;

    protected virtual void Update()
    {
        if (CurrentCooldown > 0f)
        {
            CurrentCooldown -= Time.deltaTime;
        }
    }
}

using System;
using System.Collections.Generic;
using System.Text;
using UnityEngine;


public class BlastAbility : BaseAbility
{
    [SerializeField] private Projectile _projectilePrefab;

    [SerializeField] private Transform _spawnPoint;

    [SerializeField] private float _projectileSpeed = 20f;
    [SerializeField] private float _damage = 50f;


    public override void Execute(Vector3 direction)
    {
        if (!IsReady || _projectilePrefab == null) return;


        if (direction == Vector3.zero)
        {
            direction = Owner.Transform.forward;
        }

        Vector3 origin = _spawnPoint != null
            ? _spawnPoint.position
            : Owner.Transform.position + Owner.Transform.forward * 0.5f + Vector3.up * 0.5f;

        Projectile projectile = Instantiate(_projectilePrefab, origin, Quaternion.LookRotation(direction));

        projectile.Initialize(direction * _projectileSpeed, _damage, Owner.Transform);

        StartCooldown();
    }
}

using UnityEngine;

public class ReplayVisualizer : MonoBehaviour
{
    private LineRenderer _recordedLine;
    private LineRenderer _playbackLine;

    private PlayerController _player;
    private InputRecorder _recorder;

    private bool _wasRecording;
    private bool _wasPlayingBack;

    // Shared material across all visualizer line instances to avoid material leaks.
    private static Material lineMaterial;

    {
        _player = PlayerController.Instance;
        _recorder = Object.FindFirstObjectByType<InputRecorder>();

        _recordedLine = CreateLine("Recorded Path (Blue)", Color.blue, width: 0.1f);

    private LineRenderer CreateLine(string objectName, Color color, float width)
    {
        if (lineMaterial == null)
            lineMaterial = new Material(Shader.Find("Sprites/Default"));

        var go = new GameObject(objectName);
        go.transform.SetParent(transform);

        var lr = go.AddComponent<LineRenderer>();
        lr.startWidth = width;
        lr.endWidth = width;
        lr.material = lineMaterial;
        lr.startColor = color;
        lr.endColor = color;
        lr.positionCount = 0;
        return lr;
    }

    private void Update()
    {
        if (Input.GetKeyDown(KeyCode.M))
        {
            _recordedLine.positionCount = 0;
            _playbackLine.positionCount = 0;
        }
    }

    private void FixedUpdate()
    {
        if (_player == null || _recorder == null) return;

        bool isRecordingNow = _recorder.IsRecording;
        bool isPlayingBackNow = _player.IsPlayingBack;

        if (isRecordingNow)
        {
            if (!_wasRecording) _recordedLine.positionCount = 0;
            AppendPoint(_recordedLine, _player.transform.position);
        }

        if (isPlayingBackNow)
        {
            if (!_wasPlayingBack) _playbackLine.positionCount = 0;
            AppendPoint(_playbackLine, _player.transform.position);
        }

        _wasRecording = isRecordingNow;
        _wasPlayingBack = isPlayingBackNow;
    }

    private static void AppendPoint(LineRenderer lr, Vector3 position)
    {
        lr.positionCount++;
        // Offset upward slightly so the line doesn't clip into floor geometry.
        lr.SetPosition(lr.positionCount - 1, position + Vector3.up * 0.1f);
    }

        {
            Time.timeScale = 0.1f;
            // Scale fixedDeltaTime to maintain physics determinism at reduced speed.
            Time.fixedDeltaTime = defaultFixedTimestep * Time.timeScale;
            Debug.Log("<color=yellow>[DEBUG] Slow Motion: ON</color>");
        }
        else
        {
            Time.timeScale = 1f;
            Time.fixedDeltaTime = defaultFixedTimestep;
            Debug.Log("<color=yellow>[DEBUG] Slow Motion: OFF</color>");
        }
    }

using System;
using UnityEngine;

public class HealthComponent : MonoBehaviour
{
    [SerializeField] private float _maxHealth = 100f;

    /// <summary>Fired when health changes. Args: currentHealth, maxHealth.</summary>
    public event Action<float, float> OnHealthChanged;

    /// <summary>Fired once when health reaches zero. Subscriber (RoundManager) handles round resolution.</summary>
    public event Action OnDeath;

    public float CurrentHealth { get; private set; }
    public float MaxHealth => _maxHealth;
    public bool IsDead { get; private set; }

    private void Awake()
    {
        CurrentHealth = _maxHealth;
    }

    public void TakeDamage(float amount)
    {
        if (IsDead) return;

        CurrentHealth = Mathf.Max(0f, CurrentHealth - amount);
        OnHealthChanged?.Invoke(CurrentHealth, _maxHealth);

        if (CurrentHealth <= 0f)
        {
            IsDead = true;
            OnDeath?.Invoke();
        }
    }

    public void Heal(float amount)
    {
        if (IsDead) return;

        CurrentHealth = Mathf.Min(_maxHealth, CurrentHealth + amount);
        OnHealthChanged?.Invoke(CurrentHealth, _maxHealth);
    }

    /// <summary>Call at the start of each round to restore the fighter to full health.</summary>
    public void ResetHealth()
    {
        IsDead = false;
        CurrentHealth = _maxHealth;
        OnHealthChanged?.Invoke(CurrentHealth, _maxHealth);
    }
}

using System;
using System.Linq.Expressions;
using Unity.Cinemachine;
using Unity.VisualScripting;
using UnityEngine;

[RequireComponent(typeof(Rigidbody))]
public class PlayerController : MonoBehaviour, ICharacterPhysics
{
    public static PlayerController Instance { get; private set; }

    public bool IsDeflected { get; private set; }
    public bool IsPlayingBack { get; private set; }
    public Vector3 CurrentMoveDirection { get; private set; }
    public Vector3 StartPosition => _startPosition;

    Transform ICharacterPhysics.Transform => transform;

    [Header("Movement Settings")]
    [SerializeField] private float _moveSpeed = 7f;
    [SerializeField] private float _jumpForce = 5f;
    [SerializeField] private float _dashFriction = 8f;

    [Header("Ground Detection")]
    [SerializeField] private float _playerHeight = 2f;
    [SerializeField] private LayerMask _groundLayer;

    private Vector3 _startPosition;

    private Rigidbody _rigidbody;
    private IInputSource _inputSource;

    private bool _isGrounded;
    private bool _pendingJump;
    private bool _pendingAbilityUse;
    private bool _pendingFireUse;
    private bool _forcePending;
    private bool _justRecovered;

    private Vector3 _dashVelocity;

    private AbilityManager _abilityManager;


    private void Awake()
    {
        if (Instance != null && Instance != this)
        {
            Destroy(gameObject);
            return;
        }
        Instance = this;
    }

    private void Start()
    {
        _rigidbody = GetComponent<Rigidbody>();
        _abilityManager = GetComponent<AbilityManager>();

        _rigidbody.freezeRotation = true;
        _rigidbody.interpolation = RigidbodyInterpolation.Interpolate;

        _startPosition = transform.position;
    }

    private void OnDestroy()
    {
        if (_inputSource != null)
        {
            _inputSource.OnJumpPressed -= OnJumpInput;
            _inputSource.OnMobilityAbilityPressed -= OnUseAbilityInput;
        }
    }

    public void SetInputSource(IInputSource inputSource)
    {
        if (_inputSource != null)
        {
            _inputSource.OnJumpPressed -= OnJumpInput;
            _inputSource.OnMobilityAbilityPressed -= OnUseAbilityInput;
        }
        _inputSource = inputSource;
        _inputSource.OnJumpPressed += OnJumpInput;
        _inputSource.OnMobilityAbilityPressed += OnUseAbilityInput;
        _inputSource.OnFirePressed += OnFireInput;

        if (_inputSource is PlayerInputSource)
        {
            IsPlayingBack = false;
        }
        else if (_inputSource is InputPlayback)
        {
            IsPlayingBack = true;
        }   
    }

    public IInputSource InputSource => _inputSource;

    private void OnJumpInput()
    {
        _pendingJump = true;
    }

    private void OnUseAbilityInput()
    {
        _pendingAbilityUse = true;
    }

    private void OnFireInput()
    {
        _pendingFireUse = true;
    }

    private void FixedUpdate()
    {
        UpdateGroundState();
        ResolveDeflectionRecovery();
        ResolveJump();
        HandleMovement();
    }

    private void UpdateGroundState()
    {
        Vector3 rayStart = transform.position + Vector3.down * (_playerHeight * 0.4f);
        _isGrounded = Physics.Raycast(rayStart, Vector3.down, 0.4f, _groundLayer, QueryTriggerInteraction.Ignore);
        _justRecovered = false;
    }

    private void ResolveDeflectionRecovery()
    {
        if (!IsDeflected) return;
        if (_forcePending)
        {
            _forcePending = false;
            return;
        }

        Vector2 horizontalVelocity = new Vector2(_rigidbody.linearVelocity.x, _rigidbody.linearVelocity.z);
        bool slowedDown = horizontalVelocity.magnitude <= _moveSpeed + 1f;
        bool notRocketingUp = _rigidbody.linearVelocity.y <= 1f;

        if (_isGrounded && slowedDown && notRocketingUp)
        {
            IsDeflected = false;
            _justRecovered = true;
            _rigidbody.linearVelocity = new Vector3(0f, _rigidbody.linearVelocity.y, 0f);
        }
    }

    private void ResolveJump()
    {
        if (!_pendingJump) return;
        _pendingJump = false;

        if (_isGrounded && !IsDeflected)
        {
            _rigidbody.linearVelocity = new Vector3(_rigidbody.linearVelocity.x, 0f, _rigidbody.linearVelocity.z);
            _rigidbody.AddForce(Vector3.up * _jumpForce, ForceMode.Impulse);
        }
    }

    private void HandleMovement()
    {
        if (IsDeflected || _justRecovered)
        {
            CurrentMoveDirection = Vector3.zero;
            return; 
        }

        _inputSource.Tick();

        _dashVelocity = Vector3.MoveTowards(_dashVelocity, Vector3.zero, _dashFriction * Time.fixedDeltaTime);

        Vector3 moveDirection = _inputSource.GetWorldMoveDirection();
        CurrentMoveDirection = moveDirection;

        _rigidbody.linearVelocity = new Vector3(
            moveDirection.x * _moveSpeed + _dashVelocity.x,
            _rigidbody.linearVelocity.y,
            moveDirection.z * _moveSpeed + _dashVelocity.z
        );

        //_rigidbody.linearVelocity = new Vector3(moveDirection.x * _moveSpeed, _rigidbody.linearVelocity.y, moveDirection.z * _moveSpeed);

        if (moveDirection != Vector3.zero)
        {
            Quaternion targetRotation = Quaternion.LookRotation(moveDirection);
            _rigidbody.MoveRotation(Quaternion.Slerp(_rigidbody.rotation, targetRotation, 10f * Time.fixedDeltaTime));
        }

        if (_pendingAbilityUse)
        {
            _pendingAbilityUse = false;
            Vector3 abilityDirection = moveDirection == Vector3.zero ? transform.forward : moveDirection;
            _abilityManager?.TryUseMobility(abilityDirection);
        }

        if (_pendingFireUse)
        {
            _pendingFireUse = false;
            _abilityManager?.TryUseFire(_inputSource.GetAimDirection());
        }
    }

    public void ApplyDeflectionForce(Vector3 force)
    {
        Debug.Assert(!_forcePending,
            "[PHYSICS FATAL] A new deflection force was applied before the previous one finished pending!");

        IsDeflected = true;
        _forcePending = true;
        _pendingJump = false;

        _rigidbody.linearVelocity = Vector3.zero;
        _rigidbody.AddForce(force, ForceMode.VelocityChange);
    }

    public void ApplyDash(Vector3 force)
    {
        _dashVelocity += force;
    }

    private void OnDrawGizmos()
    {
        Vector3 rayStart = transform.position + Vector3.down * (_playerHeight * 0.4f);
        Gizmos.color = _isGrounded ? Color.green : Color.red;
        Gizmos.DrawWireSphere(rayStart, 0.1f);
        Gizmos.DrawLine(rayStart, rayStart + Vector3.down * 0.4f);
    }

}

using System;
using Unity.VisualScripting;
using UnityEngine;

[DefaultExecutionOrder(-10)]
public class InputPlayback : MonoBehaviour, IInputSource
{
    public event Action OnFirePressed;
    public event Action OnJumpPressed;
    public event Action OnMobilityAbilityPressed;
    public event Action OnPlaybackCompleted;

    public enum PlaybackState { Playing, Paused, Stopped }
    public bool IsPlayback => true;
    public bool IsPlayingBack => _playbackState == PlaybackState.Playing;

    private Recording _recording;
    private int _currentFrameIndex;
    private int _lastConsumedIndex;
    private Vector2 _pendingLookDelta;
    private PlaybackState _playbackState = PlaybackState.Stopped;

    public Vector2 GetMoveInput()
    {
        if (!IsPlayingBack || _recording == null || _lastConsumedIndex >= _recording.Frames.Count)
            return Vector2.zero;

        return _recording.Frames[_lastConsumedIndex].moveInput;
    }

    public Vector3 GetWorldMoveDirection()
    {
        if (!IsPlayingBack || _recording == null || _lastConsumedIndex >= _recording.Frames.Count)
            return Vector3.zero;

        if (PlayerController.Instance.IsDeflected)
            return Vector3.zero;

        return _recording.Frames[_lastConsumedIndex].worldMoveDirection;
    }

    public Vector2 GetLookDelta()
    {
        if (!IsPlayingBack || _recording == null)
            return Vector2.zero;

        Vector2 delta = _pendingLookDelta;
        _pendingLookDelta = Vector2.zero;
        return delta;
    }

    public Vector3 GetAimDirection()
    {
        if (!IsPlayingBack || _recording == null || _lastConsumedIndex >= _recording.Frames.Count)
            return Vector3.zero;
        return _recording.Frames[_lastConsumedIndex].aimDirection;
    }

    public void Tick()
    {
        AdvanceTimeline();
    }

    public void StartPlayback(Recording recording)
    {
        _recording = recording;
        _currentFrameIndex = 0;
        _lastConsumedIndex = 0;
        _pendingLookDelta = Vector2.zero;

        Rigidbody rb = PlayerController.Instance.GetComponent<Rigidbody>();
        rb.position = recording.StartPosition;
        rb.rotation = recording.StartBodyRotation;
        rb.linearVelocity = recording.StartVelocity;
        rb.angularVelocity = recording.StartAngularVelocity;

        PlayerController.Instance
            .GetComponentInChildren<CameraTargetController>()
            .SetRotation(recording.StartCameraRotation);

        Physics.SyncTransforms();

        _playbackState = PlaybackState.Playing;
    }

    public void PausePlayback()
    {
        if (_playbackState == PlaybackState.Playing)
        {
            _playbackState = PlaybackState.Paused;
        }
    }

    public void ResumePlayback()
    {
        if (_playbackState == PlaybackState.Paused)
        {
            _playbackState = PlaybackState.Playing;
        }
    }

    public void StopPlayback()
    {
        Debug.Assert(
            _recording != null && _currentFrameIndex == _recording.Frames.Count,
            $"[PLAYBACK FATAL] Frame count mismatch! Expected {_recording?.Frames.Count}, stopped at {_currentFrameIndex}.");

        _playbackState = PlaybackState.Stopped;
    }

    private void AdvanceTimeline()
    {
        if (!IsPlayingBack || _recording == null || _currentFrameIndex >= _recording.Frames.Count)
            return;


        Debug.Assert(!PlayerController.Instance.IsDeflected,
            "[PLAYBACK FATAL] The timeline advanced while the player was in the Deflected state!");

        FrameInput currentFrame = _recording.Frames[_currentFrameIndex];

        _pendingLookDelta += currentFrame.lookDelta;

        if (currentFrame.jumpPressed)
        {
            OnJumpPressed?.Invoke();
        }

        if (currentFrame.abilityPressed)
        {
            OnMobilityAbilityPressed?.Invoke();
        }
        
        if (currentFrame.firePressed)
        {
            OnFirePressed?.Invoke();
        }

        _lastConsumedIndex = _currentFrameIndex;
        _currentFrameIndex++;

        if (_currentFrameIndex >= _recording.Frames.Count)
        {
            StopPlayback();
            OnPlaybackCompleted?.Invoke();
        }
    }

}

using System;
using System.Collections.Generic;
using System.Text;
using UnityEngine;


public class RecordingManager : MonoBehaviour
{
    public static RecordingManager Instance { get; private set; }

    [SerializeField] private InputPlayback _inputPlayback;
    [SerializeField] private PlayerInputSource _playerInputSource;
    [SerializeField] private InputRecorder _inputRecorder;

    private void Awake()
    {
        if (Instance != null && Instance != this)
        {
            Destroy(gameObject);
            return;
        }
        Instance = this;
    }

    private void Start()
    {
        PlayerController.Instance.SetInputSource(_playerInputSource);

        GameInput.Instance.OnRecordAction += OnRecordStart;
        GameInput.Instance.OnRecordStopAction += OnRecordStop;
        GameInput.Instance.OnPlaybackStopAction += OnPlaybackStop;
        GameInput.Instance.OnPlaybackPauseAction += OnPlaybackPause;

        _inputPlayback.OnPlaybackCompleted += OnPlaybackFinished;
    }

    private void OnDestroy()
    {
        if (GameInput.Instance != null)
        {
            GameInput.Instance.OnRecordAction -= OnRecordStart;
            GameInput.Instance.OnRecordStopAction -= OnRecordStop;
            GameInput.Instance.OnPlaybackStopAction -= OnPlaybackStop;
            GameInput.Instance.OnPlaybackPauseAction -= OnPlaybackPause;
        }

        _inputPlayback.OnPlaybackCompleted -= OnPlaybackFinished;
    }

    private void OnRecordStart()
    {
        PlayerController.Instance.SetInputSource(_playerInputSource);
        _inputRecorder.StartRecording();
    }

    private void OnRecordStop()
    {
        Recording recording = _inputRecorder.StopRecording();
        _inputPlayback.StartPlayback(recording);

        PlayerController.Instance.SetInputSource(_inputPlayback);
    }

    private void OnPlaybackStop()
    {
        _inputPlayback.StopPlayback();
        PlayerController.Instance.SetInputSource(_playerInputSource);
    }

    private void OnPlaybackFinished()

public class CameraTargetController : MonoBehaviour
{
    [Header("Look Settings")]
    [SerializeField] private float lookSensitivity = 0.5f;
    [SerializeField] private float topClamp = -70f;
    [SerializeField] private float bottomClamp = 70f;

    private float pitch; // Up/Down (X axis)
    private float yaw;   // Left/Right (Y axis)


    public void SetRotation(Vector3 angles)
    {
        yaw = angles.y;
        pitch = angles.x;

        transform.rotation = Quaternion.Euler(angles);
    }
    public Vector3 GetCurrentRotation()
    {
        return new Vector3(pitch, yaw, 0f);
    }

    private void Update()
    {
        Vector2 lookInput = PlayerController.Instance.InputSource.GetLookDelta();

        yaw += lookInput.x * lookSensitivity;
        pitch -= lookInput.y * lookSensitivity;

        pitch = Mathf.Clamp(pitch, topClamp, bottomClamp);

        transform.rotation = Quaternion.Euler(pitch, yaw, 0f);
    }
}

using System;
using UnityEngine;
using UnityEngine.InputSystem;

public class GameInput : MonoBehaviour
{
    public static GameInput Instance { get; private set; }

    private const string PLAYER_PREFS_BINDINGS = "InputBindings";

    private InputSystem_Actions inputActions;

    public event Action OnPauseAction;
    public event Action OnBindingRebinding;
    public event Action OnJumpAction;
    public event Action OnRecordAction;
    public event Action OnRecordStopAction;
    public event Action OnPlaybackStopAction;
    public event Action OnPlaybackPauseAction;
    public event Action OnPreviousAbilityAction;
    public event Action OnNextAbilityAction;
    public event Action OnUseAbilityAction;
    public event Action OnFireAction;

    public enum Binding
    {
        MoveUp,
        MoveDown,
        MoveLeft,
        MoveRight,
        Jump,
        Pause,
        GamepadPause
    }

    private void Awake()
    {
        if (Instance != null && Instance != this)
        {
            Destroy(gameObject);
            return;
        }
        Instance = this;

        inputActions = new InputSystem_Actions();

        if (PlayerPrefs.HasKey(PLAYER_PREFS_BINDINGS))
            inputActions.LoadBindingOverridesFromJson(PlayerPrefs.GetString(PLAYER_PREFS_BINDINGS));

        inputActions.Player.Enable();

        inputActions.Player.Jump.performed += OnJump_performed;
        inputActions.Player.RecordStart.performed += OnRecordStart_performed;
        inputActions.Player.RecordFinish.performed += OnRecordStop_performed;
        inputActions.Player.PlaybackStop.performed += OnPlaybackStop_performed;
        inputActions.Player.PlaybackPause.performed += OnPlaybackPause_performed;
        inputActions.Player.PreviousAbility.performed += OnPreviousAbility_performed;
        inputActions.Player.NextAbility.performed += OnNextAbility_performed;
        inputActions.Player.UseAbility.performed += OnUseAbility_performed;
        inputActions.Player.Fire.performed += OnFire_performed;
    }

    private void OnJump_performed(InputAction.CallbackContext ctx) => OnJumpAction?.Invoke();
    private void OnRecordStart_performed(InputAction.CallbackContext ctx) => OnRecordAction?.Invoke();
    private void OnRecordStop_performed(InputAction.CallbackContext ctx) => OnRecordStopAction?.Invoke();
    private void OnPlaybackStop_performed(InputAction.CallbackContext ctx) => OnPlaybackStopAction?.Invoke();
    private void OnPlaybackPause_performed(InputAction.CallbackContext ctx) => OnPlaybackPauseAction?.Invoke();
    private void OnPreviousAbility_performed(InputAction.CallbackContext ctx) => OnPreviousAbilityAction?.Invoke();
    private void OnNextAbility_performed(InputAction.CallbackContext ctx) => OnNextAbilityAction?.Invoke();
    private void OnUseAbility_performed(InputAction.CallbackContext ctx) => OnUseAbilityAction?.Invoke();
    private void OnFire_performed(InputAction.CallbackContext ctx) => OnFireAction?.Invoke();

    public Vector2 GetMovementVectorNormalized()
    {
        return inputActions.Player.Move.ReadValue<Vector2>().normalized;
    }

    public Vector2 GetLookVector()
    {
        return inputActions.Player.Look.ReadValue<Vector2>();
    }

    private void OnDestroy()
    {
        if (inputActions == null) return;

        inputActions.Player.Jump.performed -= OnJump_performed;
        inputActions.Player.RecordStart.performed -= OnRecordStart_performed;
        inputActions.Player.RecordFinish.performed -= OnRecordStop_performed;
        inputActions.Player.PlaybackStop.performed -= OnPlaybackStop_performed;
        inputActions.Player.PlaybackPause.performed -= OnPlaybackPause_performed;
        inputActions.Player.PreviousAbility.performed -= OnPreviousAbility_performed;
        inputActions.Player.NextAbility.performed -= OnNextAbility_performed;
        inputActions.Player.UseAbility.performed -= OnUseAbility_performed;
        inputActions.Player.Fire.performed -= OnFire_performed; 
        inputActions.Player.Disable();
        inputActions.Dispose();

        OnPauseAction = null;
        OnBindingRebinding = null;
        OnJumpAction = null;
        OnRecordAction = null;
        OnRecordStopAction = null;
        OnPlaybackStopAction = null;
        OnPlaybackPauseAction = null;
        OnPreviousAbilityAction = null;
        OnNextAbilityAction = null;
        OnUseAbilityAction = null;
        OnFireAction = null;
    }

    public string GetBindingText(Binding binding)
    {
        return binding switch
        {
            Binding.MoveUp => inputActions.Player.Move.bindings[1].ToDisplayString(),
            Binding.MoveDown => inputActions.Player.Move.bindings[3].ToDisplayString(),
            Binding.MoveLeft => inputActions.Player.Move.bindings[5].ToDisplayString(),
            Binding.MoveRight => inputActions.Player.Move.bindings[7].ToDisplayString(),
            Binding.Jump => inputActions.Player.Jump.bindings[0].ToDisplayString(),
            _ => string.Empty
        };
    }

    public void RebindBinding(Binding binding, Action onComplete = null)
    {
        inputActions.Player.Disable();

        InputActionRebindingExtensions.RebindingOperation rebindOp = binding switch
        {
            Binding.MoveUp => inputActions.Player.Move.PerformInteractiveRebinding(1),
            Binding.MoveDown => inputActions.Player.Move.PerformInteractiveRebinding(3),
            Binding.MoveLeft => inputActions.Player.Move.PerformInteractiveRebinding(5),
            Binding.MoveRight => inputActions.Player.Move.PerformInteractiveRebinding(7),
            Binding.Jump => inputActions.Player.Jump.PerformInteractiveRebinding(0),
            _ => null
        };

        rebindOp?
            .OnComplete(callback =>
            {
                inputActions.Player.Enable();
                onComplete?.Invoke();
                callback.Dispose();
                PlayerPrefs.SetString(PLAYER_PREFS_BINDINGS, inputActions.SaveBindingOverridesAsJson());
                PlayerPrefs.Save();
                OnBindingRebinding?.Invoke();
            })
            .OnCancel(callback =>
            {
                inputActions.Player.Enable();
                callback.Dispose();
            })
            .Start();
    }
}

using System;

public class PlayerInputSource : MonoBehaviour, IInputSource
{
    public event Action OnFirePressed;
    public event Action OnJumpPressed;
    public event Action OnMobilityAbilityPressed;
    public bool IsPlayback => false;
    [SerializeField] private Transform _cameraTarget;

    private void Start()
    {
        GameInput.Instance.OnFireAction += HandleFire;
        GameInput.Instance.OnJumpAction += HandleJump;
        GameInput.Instance.OnUseAbilityAction += HandleMobilityAbility;
    }

    private void OnDestroy()
    {
        if (GameInput.Instance != null)
        {
            GameInput.Instance.OnFireAction -= HandleFire;
            GameInput.Instance.OnJumpAction -= HandleJump;
            GameInput.Instance.OnUseAbilityAction -= HandleMobilityAbility;
        }
    }

    private void HandleFire()
    {
        OnFirePressed?.Invoke();
    }
    private void HandleJump()
    {
        OnJumpPressed?.Invoke();
    }

    private void HandleMobilityAbility()
    {
        OnMobilityAbilityPressed?.Invoke();
    }

    public Vector3 GetAimDirection()
    {
        return _cameraTarget.forward;
    }
    public Vector2 GetMoveInput()
    {
        return GameInput.Instance.GetMovementVectorNormalized();
    }

    public Vector3 GetWorldMoveDirection()
    {
        Vector2 input = GameInput.Instance.GetMovementVectorNormalized();
        if (input == Vector2.zero) return Vector3.zero;
        
        Vector3 forward = _cameraTarget.forward;
        forward.y = 0f;
        forward.Normalize();

        Vector3 right = _cameraTarget.right;
        right.y = 0f;
        right.Normalize();

        return (forward * input.y) + (right * input.x); 
    }

    public Vector2 GetLookDelta()
    {
        return GameInput.Instance.GetLookVector();
    }

    public void Tick() { }
}

using System;
using System.Collections.Generic;
using System.Text;
using UnityEngine;

public class Projectile : MonoBehaviour
{
    [SerializeField] private float _lifetime = 5f;

    private float _damage;
    private Transform _ownerRoot;
    private Rigidbody _rigidbody;

    private void Awake()
    {
        _rigidbody = GetComponent<Rigidbody>();
    }

    public void Initialize(Vector3 velocity, float damage, Transform ownerRoot)
    {
        _damage = damage;
        _ownerRoot = ownerRoot;

        _rigidbody.linearVelocity = velocity;

        Destroy(gameObject, _lifetime);
    }

    private void OnTriggerEnter(Collider other)
    {
        if (_ownerRoot != null && other.transform.IsChildOf(_ownerRoot))
            return;

        if (other.TryGetComponent<HealthComponent>(out var health))
        {
            health.TakeDamage(_damage);
        }

        Explode();
    }

    private void Explode()
    {
        Destroy(gameObject);
    }
}
