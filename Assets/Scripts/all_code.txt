using System;
using UnityEngine;

// Runs before PlayerController so GetMoveDirection() is always current-frame data
[DefaultExecutionOrder(-10)]
public class InputPlayback : MonoBehaviour, IInputSource
{
    public event Action OnJumpPressed;

    private Recording recording;
    private int currentFrameIndex = 0;
    private int lastConsumedIndex = 0;
    private float playbackStartTime;
    private bool isPlayingBack = false;

    private Vector2 pendingLookDelta = Vector2.zero;

    public bool IsPlayingBack => isPlayingBack;

    public Vector2 GetMoveDirection()
    {
        if (!isPlayingBack || recording == null || lastConsumedIndex >= recording.Frames.Count)
            return Vector2.zero;
        return recording.Frames[lastConsumedIndex].moveDirection;
    }
    public Vector3 GetWorldMoveDirection()
    {
        if (!isPlayingBack || recording == null || lastConsumedIndex >= recording.Frames.Count)
            return Vector3.zero;
        return recording.Frames[lastConsumedIndex].worldMoveDirection;
    }

    public Vector2 GetLookDelta()
    {
        if (!isPlayingBack || recording == null)
            return Vector2.zero;

        Vector2 delta = pendingLookDelta;
        pendingLookDelta = Vector2.zero; // drain — next Update call gets zero
        return delta;
    }

    public void StartPlayback(Recording recording)
    {
        this.recording = recording;
        currentFrameIndex = 0;
        lastConsumedIndex = 0;
        playbackStartTime = Time.time;
        pendingLookDelta = Vector2.zero;

        Rigidbody rb = PlayerController.Instance.GetComponent<Rigidbody>();
        rb.linearVelocity = Vector3.zero;
        rb.angularVelocity = Vector3.zero;
        rb.position = recording.StartPosition;

        PlayerController.Instance.transform.GetComponentInChildren<CameraTargetController>().SetRotation(recording.StartRotation);

        Physics.SyncTransforms();
        PlayerController.Instance.SetInputSource(this);
        isPlayingBack = true;
    }

    public void StopPlayback()
    {
        isPlayingBack = false;
    }

    private void FixedUpdate()
    {
        if (!isPlayingBack || recording == null || currentFrameIndex >= recording.Frames.Count)
            return;

        // Grab the exact frame for this specific physics tick
        FrameInput currentFrame = recording.Frames[currentFrameIndex];

        pendingLookDelta += currentFrame.lookDelta;

        if (currentFrame.jumpPressed)
            OnJumpPressed?.Invoke();

        // Store the index so GetMoveDirection() reads the correct data
        lastConsumedIndex = currentFrameIndex;

        // Advance the tick by exactly 1
        currentFrameIndex++;

        if (currentFrameIndex >= recording.Frames.Count)
        {
            StopPlayback();
            Debug.Log("Playback finished.");
        }
    }
}

using System;
using System.Collections.Generic;
using UnityEngine;

public class InputRecorder : MonoBehaviour
{
    private List<FrameInput> inputFrames = new();
    private bool isRecording = false;
    private bool jumpPressedThisFixedFrame = false;
    private Vector2 accumulatedLookDelta = Vector2.zero;

    private Vector3 recordingStartPosition;
    private Vector3 recordingStartRotation;

    private void Start()
    {
        GameInput.Instance.OnJumpAction += HandleJump;
    }

    private void OnDestroy()
    {
        if (GameInput.Instance != null)
            GameInput.Instance.OnJumpAction -= HandleJump;
    }

    private void HandleJump(object sender, EventArgs e)
    {
        if (isRecording)
            jumpPressedThisFixedFrame = true;
    }

    private void Update()
    {
        if (!isRecording) return;
        accumulatedLookDelta += GameInput.Instance.GetLookVector(); // accumulate every Update tick
    }

    public void StartRecording()
    {
        inputFrames.Clear();
        accumulatedLookDelta = Vector2.zero;
        isRecording = true;
        recordingStartPosition = PlayerController.Instance.transform.position;
        recordingStartRotation = PlayerController.Instance.transform.GetComponentInChildren<CameraTargetController>().GetCurrentRotation();
    }

    public Recording StopRecording()
    {
        isRecording = false;
        return new Recording(
            inputFrames,
            recordingStartPosition,
            recordingStartRotation
        );
    }

    private void FixedUpdate()
    {
        if (!isRecording) return;

        inputFrames.Add(new FrameInput
        {
            moveDirection = GameInput.Instance.GetMovementVectorNormalized(),

            // 2. GRAB THE EXACT 3D VECTOR from the PlayerController!
            worldMoveDirection = PlayerController.Instance.CurrentMoveDirection,

            lookDelta = accumulatedLookDelta,
            jumpPressed = jumpPressedThisFixedFrame,
            firePressed = false,
            abilityPressed = false
        });

        accumulatedLookDelta = Vector2.zero;
        jumpPressedThisFixedFrame = false;
    }
}

    {
        Recording recording = inputRecorder.StopRecording();
        inputPlayback.StartPlayback(recording);
    }

using UnityEngine;

public class CameraTargetController : MonoBehaviour
{
    [Header("Look Settings")]
    [SerializeField] private float lookSensitivity = 0.5f;
    [SerializeField] private float topClamp = -70f;
    [SerializeField] private float bottomClamp = 70f;

    private float pitch; // Up/Down (X axis)
    private float yaw;   // Left/Right (Y axis)


    public void SetRotation(Vector3 angles)
    {
        yaw = angles.y;
        pitch = angles.x;

        transform.rotation = Quaternion.Euler(angles);
    }
    public Vector3 GetCurrentRotation()
    {
        return new Vector3(pitch, yaw, 0f);
    }

    private void Update()
    {
        Vector2 lookInput = PlayerController.Instance.InputSource.GetLookDelta();

        yaw += lookInput.x * lookSensitivity;
        pitch -= lookInput.y * lookSensitivity;

        pitch = Mathf.Clamp(pitch, topClamp, bottomClamp);

        transform.rotation = Quaternion.Euler(pitch, yaw, 0f);
    }
}

using System;
using UnityEngine;
using UnityEngine.InputSystem;

public class GameInput : MonoBehaviour
{
    private const string PLAYER_PREFS_BINDINGS = "InputBindings";

    private InputSystem_Actions inputAction;

    public event EventHandler OnPauseAction;
    public event EventHandler OnBindingRebinding;

    public event EventHandler OnJumpAction;
    public event EventHandler OnRecordAction;
    public event EventHandler OnRecordStopAction;
    public event EventHandler OnPlaybackStopAction;
    public bool IsJumpPressed => inputAction.Player.Jump.triggered;

    public enum Binding
    {
        MoveUp,
        MoveDown,
        MoveLeft,
        MoveRight,
        Jump,
        Pause,

        GamepadPause
    }
    public static GameInput Instance { get; private set; }

    private void Awake()
    {
        if (Instance != null && Instance != this)
        {
            Destroy(gameObject);
            return;
        }
        Instance = this;
        inputAction = new InputSystem_Actions();

        if (PlayerPrefs.HasKey(PLAYER_PREFS_BINDINGS))
        {
            inputAction.LoadBindingOverridesFromJson(PlayerPrefs.GetString(PLAYER_PREFS_BINDINGS));
        }

        inputAction.Player.Enable();

        inputAction.Player.Jump.performed += Jump_performed;
        inputAction.Player.RecordStart.performed += RecordStart_performed;
        inputAction.Player.RecordFinish.performed += RecordStop_performed;
        inputAction.Player.PlaybackStop.performed += PlaybackStop_performed;
    }

    private void Pause_performed(UnityEngine.InputSystem.InputAction.CallbackContext obj)
    {
        OnPauseAction?.Invoke(this, EventArgs.Empty);
    }

    private void RecordStop_performed(UnityEngine.InputSystem.InputAction.CallbackContext obj)
    {
        OnRecordStopAction?.Invoke(this, EventArgs.Empty);
    }

    private void RecordStart_performed(UnityEngine.InputSystem.InputAction.CallbackContext obj)
    {
        OnRecordAction?.Invoke(this, EventArgs.Empty);
    }

    private void PlaybackStop_performed(UnityEngine.InputSystem.InputAction.CallbackContext obj)
    {
        OnPlaybackStopAction?.Invoke(this, EventArgs.Empty);
    }
    private void Jump_performed(UnityEngine.InputSystem.InputAction.CallbackContext obj)
    {
        OnJumpAction?.Invoke(this, EventArgs.Empty);
    }

    public Vector2 GetMovementVectorNormalized()
    {
        Vector2 inputVector = inputAction.Player.Move.ReadValue<Vector2>();

        inputVector = inputVector.normalized;

        return inputVector;
    }

    public Vector2 GetLookVector()
    {
        return inputAction.Player.Look.ReadValue<Vector2>();
    }

    private void OnDestroy()
    {
        OnPauseAction = null;

        if (inputAction != null)
        {
            inputAction.Player.Jump.performed -= Jump_performed;

            inputAction.Player.Disable();
            inputAction.Dispose();
        }

        OnPauseAction = null;
        OnJumpAction = null; 
    }

    public string GetBindingText(Binding binding)
    {
        switch (binding)
        {
            default:

            case Binding.MoveUp:
                return inputAction.Player.Move.bindings[1].ToDisplayString();

            case Binding.MoveDown:
                return inputAction.Player.Move.bindings[3].ToDisplayString();

            case Binding.MoveLeft:
                return inputAction.Player.Move.bindings[5].ToDisplayString();

            case Binding.MoveRight:
                return inputAction.Player.Move.bindings[7].ToDisplayString();

            case Binding.Jump:
                return inputAction.Player.Jump.bindings[0].ToDisplayString();
        }
    }

    public void RebindBinding(Binding binding, Action OnComplete = null)
    {
        inputAction.Player.Disable();

        InputActionRebindingExtensions.RebindingOperation rebindingOperation = null;

        switch (binding)
        {
            default:
            case Binding.MoveUp:
                rebindingOperation = inputAction.Player.Move.PerformInteractiveRebinding(1);
                break;
            case Binding.MoveDown:
                rebindingOperation = inputAction.Player.Move.PerformInteractiveRebinding(3);
                break;
            case Binding.MoveLeft:
                rebindingOperation = inputAction.Player.Move.PerformInteractiveRebinding(5);
                break;
            case Binding.MoveRight:
                rebindingOperation = inputAction.Player.Move.PerformInteractiveRebinding(7);
                break;

            case Binding.Jump:
                rebindingOperation = inputAction.Player.Jump.PerformInteractiveRebinding(0);
                break;
        }

        rebindingOperation?
            .OnComplete((callback) =>
            {
                inputAction.Player.Enable();
                OnComplete?.Invoke();
                callback.Dispose();

                PlayerPrefs.SetString(PLAYER_PREFS_BINDINGS, inputAction.SaveBindingOverridesAsJson());
                PlayerPrefs.Save();
                OnBindingRebinding?.Invoke(this, EventArgs.Empty);
            })
            .OnCancel((callback) =>
            {
                inputAction.Player.Enable();
                callback.Dispose();
            })
            .Start();
    }

}

using System;
using Unity.Cinemachine;
using Unity.VisualScripting;
using UnityEngine;

[RequireComponent(typeof(Rigidbody))]
public class PlayerController : MonoBehaviour
{
    public static PlayerController Instance { get; private set; }

    [Header("Movement Settings")]
    [SerializeField] private float moveSpeed = 7f;
    [SerializeField] private float jumpForce = 5f;

    [Header("Ground Detection")]
    [SerializeField] private float playerHeight = 2f;
    [SerializeField] private LayerMask groundLayer;

    [Header("References")]
    [SerializeField] private Transform cameraTarget; 


    private Rigidbody rb;
    private IInputSource inputSource;
    private bool isGrounded;

    private Vector3 startPosition;
    private Vector3 startRotation;

    private bool isPlaybackActive = false;

    private Vector3 cachedForward;
    private Vector3 cachedRight;

    private bool pendingJump = false;

    public bool IsPlayingBack => isPlaybackActive;

    public Vector3 StartPosition => startPosition;
    public Vector3 StartRotation => startRotation;

    public IInputSource InputSource => inputSource;
    public Vector3 CurrentMoveDirection { get; private set; }
    private void Awake()
    {
        if (Instance != null && Instance != this)
        {
            Destroy(gameObject);
            return;
        }
        Instance = this;
    }

    public void SetInputSource(IInputSource source)
    {
        if (inputSource != null)
            inputSource.OnJumpPressed -= GameInput_OnJumpAction;

        isPlaybackActive = source is InputPlayback;
        inputSource = source;
        inputSource.OnJumpPressed += GameInput_OnJumpAction;
    }

    private void Start()
    {
        rb = GetComponent<Rigidbody>();
        rb.freezeRotation = true;
        rb.interpolation = RigidbodyInterpolation.Interpolate; 

        startPosition = transform.position;
        startRotation = transform.eulerAngles;
    }

    private void Update()
    {
        cachedForward = cameraTarget.forward;
        cachedForward.y = 0f;
        cachedForward.Normalize();

        cachedRight = cameraTarget.right;
        cachedRight.y = 0f;
        cachedRight.Normalize();
    }

    private void FixedUpdate()
    {
        // 1. Calculate physics states ONLY in the physics step
        isGrounded = Physics.Raycast(transform.position, Vector3.down, (playerHeight * 0.5f) + 0.2f, groundLayer);

        // 2. Safely apply jump forces
        if (pendingJump)
        {
            if (isGrounded)
            {
                rb.linearVelocity = new Vector3(rb.linearVelocity.x, 0f, rb.linearVelocity.z);
                rb.AddForce(Vector3.up * jumpForce, ForceMode.Impulse);
            }
            pendingJump = false; // Drain the jump
        }

        // 3. Move
        HandleMovement();
    }
    public void HandleMovement()
    {
        Vector3 moveDirection;

        if (isPlaybackActive)
        {
            // BYPASS the fluctuating camera entirely! 
            // Use the exact 3D vector captured during the recording tick.
            moveDirection = inputSource.GetWorldMoveDirection();
        }
        else
        {
            Vector2 inputVector = inputSource.GetMoveDirection();
            moveDirection = (cachedForward * inputVector.y) + (cachedRight * inputVector.x);
        }

        // Save it so the InputRecorder can grab it!
        CurrentMoveDirection = moveDirection;

        rb.linearVelocity = new Vector3(
            moveDirection.x * moveSpeed,
            rb.linearVelocity.y,
            moveDirection.z * moveSpeed
        );

        if (moveDirection != Vector3.zero)
        {
            Quaternion targetRotation = Quaternion.LookRotation(moveDirection);
            rb.MoveRotation(Quaternion.Slerp(rb.rotation, targetRotation, 10f * Time.fixedDeltaTime));
        }
    }

    private void GameInput_OnJumpAction()
    {
        pendingJump = true;
    }

    private void OnDestroy()
    {
        if (GameInput.Instance != null)
        {
            inputSource.OnJumpPressed -= GameInput_OnJumpAction;
        }
    }
}
